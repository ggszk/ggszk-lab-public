    Lesson 8 演習解答¶

    演習1: 解答例¶

    import numpy as np
from audio_lib import sawtooth_wave, adsr, AudioSignal
from audio_lib.notebook import play_sound
from IPython.display import display

dur = 2.0
freq = 262  # C4
wave = sawtooth_wave(freq, dur)

# 1. フルート風
env_flute = adsr(dur, attack=0.3, decay=0.1, sustain=0.7, release=0.4)
flute = AudioSignal(wave.data * env_flute.data, 44100)
display(play_sound(flute, "フルート風"))

# 2. マリンバ風
env_marimba = adsr(dur, attack=0.005, decay=0.3, sustain=0.1, release=0.05)
marimba = AudioSignal(wave.data * env_marimba.data, 44100)
display(play_sound(marimba, "マリンバ風"))

# 3. ストリングスのパッド
env_pad = adsr(dur, attack=0.8, decay=0.1, sustain=0.9, release=0.5)
pad = AudioSignal(wave.data * env_pad.data, 44100)
display(play_sound(pad, "ストリングスパッド風"))

    パラメータの理由:

    1. フルート風: 息を吹き込んで徐々に音が出るため A が長め（0.3）。安定して鳴るため S が高め（0.7）。息を止めた後もゆっくり消える（R=0.4）。
    2. マリンバ風: バチで叩くので A は極めて短い（0.005）。木の共鳴はすぐ減衰する（D=0.3）。持続音はほぼない（S=0.1）。叩いた瞬間の音が主体で余韻は短い（R=0.05）。
    3. ストリングスパッド: 多数の弦がゆっくり立ち上がる（A=0.8）。安定して鳴り続ける（S=0.9）。弓を離しても共鳴がゆっくり消える（R=0.5）。

    演習2: 解答例¶

    import matplotlib.pyplot as plt
import numpy as np
from audio_lib import adsr, AudioSignal

dur = 2.0
params = [
    (0.5, 0.0, 1.0, 0.5),
    (0.01, 0.3, 0.3, 0.2),
    (0.01, 0.0, 1.0, 0.01),
    (0.2, 0.2, 0.5, 0.5),
]

fig, axes = plt.subplots(2, 2, figsize=(12, 6))

for ax, (a, d, s, r) in zip(axes.flat, params):
    env = adsr(dur, attack=a, decay=d, sustain=s, release=r)
    t = np.arange(len(env.data)) / 44100
    ax.plot(t, env.data, linewidth=2)
    ax.set_title(f"A={a}, D={d}, S={s}, R={r}")
    ax.set_ylim(-0.05, 1.1)
    ax.set_xlabel("時間 (秒)")
    ax.set_ylabel("レベル")
    ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

    記述例: 1. (A=0.5, D=0, S=1.0, R=0.5): ゆっくり立ち上がり、最大音量で持続し、ゆっくり消える。弦楽器やパッドシンセに近い。 2. (A=0.01, D=0.3, S=0.3, R=0.2): 素早い立ち上がりと減衰。ピアノやギターのような撥弦楽器的。 3. (A=0.01, D=0, S=1.0, R=0.01): 瞬時にオン・オフ。オルガンや電子音的。矩形的なエンベロープ。 4. (A=0.2, D=0.2, S=0.5, R=0.5): バランスの取れた変化。木管楽器的な穏やかな音。

    演習3: 解答例¶

    import numpy as np
from audio_lib import sine_wave, sawtooth_wave, adsr, note_to_frequency, AudioSignal
from audio_lib.notebook import play_sound
from IPython.display import display

c_major = [60, 62, 64, 65, 67, 69, 71, 72]
dur_note = 0.5

# 1. エンベロープなし（サイン波）
parts_raw = []
for midi_num in c_major:
    freq = note_to_frequency(midi_num)
    sig = sine_wave(freq, dur_note)
    parts_raw.append(sig.data)
raw_signal = AudioSignal(np.concatenate(parts_raw), 44100)
display(play_sound(raw_signal, "1. サイン波（エンベロープなし）"))

# 2. エンベロープあり（サイン波）
parts_env = []
for midi_num in c_major:
    freq = note_to_frequency(midi_num)
    wave = sine_wave(freq, dur_note)
    env = adsr(dur_note, attack=0.02, decay=0.1, sustain=0.6, release=0.1)
    shaped = AudioSignal(wave.data * env.data, 44100)
    parts_env.append(shaped.data)
env_signal = AudioSignal(np.concatenate(parts_env), 44100)
display(play_sound(env_signal, "2. サイン波 + ADSR"))

# 3. ノコギリ波 + ピアノ風エンベロープ
parts_piano = []
for midi_num in c_major:
    freq = note_to_frequency(midi_num)
    wave = sawtooth_wave(freq, dur_note)
    env = adsr(dur_note, attack=0.01, decay=0.2, sustain=0.3, release=0.1)
    shaped = AudioSignal(wave.data * env.data, 44100)
    parts_piano.append(shaped.data)
piano_signal = AudioSignal(np.concatenate(parts_piano), 44100)
display(play_sound(piano_signal, "3. ノコギリ波 + ピアノ風 ADSR"))

    記述例: - 1（エンベロープなし）は各音の境界で「プチッ」というクリックノイズが聞こえ、機械的で不自然。 - 2（サイン波 + ADSR）はクリックノイズが消え、各音が自然に立ち上がり、減衰する。ただしサイン波なので音色は素朴。 - 3（ノコギリ波 + ピアノ風）は倍音の豊かさとピアノ的な減衰が組み合わさり、最も楽器らしい。波形（音色）とエンベロープ（表情）の両方が音の印象を決めていることがわかる。

    演習4: 解答例¶

    import numpy as np
import matplotlib.pyplot as plt
from audio_lib import sine_wave, adsr, AudioSignal
from audio_lib.notebook import play_sound
from IPython.display import display

dur = 1.5
sustain_levels = [0.0, 0.2, 0.5, 0.8, 1.0]

# 1-2. グラフに重ねて描画
plt.figure(figsize=(10, 5))
for s in sustain_levels:
    env = adsr(dur, attack=0.05, decay=0.2, sustain=s, release=0.3)
    t = np.arange(len(env.data)) / 44100
    plt.plot(t, env.data, linewidth=2, label=f"S={s}")

plt.xlabel("時間 (秒)")
plt.ylabel("レベル")
plt.title("Sustain レベルの違い（A=0.05, D=0.2, R=0.3）")
plt.legend()
plt.grid(True, alpha=0.3)
plt.ylim(-0.05, 1.1)
plt.tight_layout()
plt.show()

# 3. 音を鳴らして聞き比べ
wave = sine_wave(440, dur)
for s in sustain_levels:
    env = adsr(dur, attack=0.05, decay=0.2, sustain=s, release=0.3)
    shaped = AudioSignal(wave.data * env.data, 44100)
    display(play_sound(shaped, f"S={s}"))

    記述例: - S=0.0: ディケイ後に音量がゼロになる。パーカッション的で短い音。リリースも音量がゼロから始まるため聞こえない。 - S=0.2: ディケイ後にかすかに鳴り続ける。ピアノや木琴のような減衰楽器に近い。 - S=0.5: アタックの半分の音量で持続。バランスの良い音。 - S=0.8: ほぼアタックと同じ音量で持続。管楽器的。 - S=1.0: ディケイの効果が消える（1.0 → 1.0 は変化なし）。アタック後ずっと最大音量。オルガン的。

    演習5: 解答例¶

    本文 8.6 節で定義した make_adsr_linear 関数を使います。

    import numpy as np
import matplotlib.pyplot as plt
from audio_lib import adsr, sawtooth_wave, AudioSignal
from audio_lib.notebook import play_sound
from IPython.display import display

dur = 1.5
a, d, s, r = 0.1, 0.3, 0.4, 0.3

# エンベロープを生成（make_adsr_linear は本文 8.6 節で定義）
env_lin = make_adsr_linear(dur, a, d, s, r)
env_exp = adsr(dur, attack=a, decay=d, sustain=s, release=r)

# 1. グラフに重ねて描画
plt.figure(figsize=(10, 5))
t_lin = np.arange(len(env_lin.data)) / 44100
t_exp = np.arange(len(env_exp.data)) / 44100
plt.plot(t_lin, env_lin.data, linewidth=2, label="リニア（自作）")
plt.plot(t_exp, env_exp.data, linewidth=2, label="指数カーブ（audio_lib）")
plt.xlabel("時間 (秒)")
plt.ylabel("レベル")
plt.title(f"リニア vs 指数カーブ（A={a}, D={d}, S={s}, R={r}）")
plt.legend()
plt.grid(True, alpha=0.3)
plt.ylim(-0.05, 1.1)
plt.tight_layout()
plt.show()

# 2. 聞き比べ
wave = sawtooth_wave(440, dur)
shaped_lin = AudioSignal(wave.data * env_lin.data, 44100)
shaped_exp = AudioSignal(wave.data * env_exp.data, 44100)

display(play_sound(shaped_lin, "リニアエンベロープ"))
display(play_sound(shaped_exp, "指数カーブエンベロープ"))

    記述例: - アタック: 指数カーブは立ち上がりの初動が速く、リニアよりシャープに聞こえる。リニアは一定の速さでゆっくり立ち上がる。 - ディケイ: 指数カーブは最初に急速に下がり、サステインレベル付近でゆるやかになる。リニアは一定の速さで下がる。指数のほうが「自然な減衰」に聞こえる。 - リリース: 指数カーブは最初に素早く減衰し、最後は余韻がゆっくり消える。リニアは直線的に消える。指数のほうが実際の楽器の残響に近い。 - 全体的に、指数カーブのほうがアナログシンセサイザーや実際の楽器の挙動に近く、自然な印象を与える。

    演習6: 解答例¶

    関数の実装¶

    import numpy as np
from audio_lib import (sine_wave, sawtooth_wave, square_wave, triangle_wave,
                       adsr, AudioSignal, note_to_frequency)
from audio_lib.notebook import play_sound
from IPython.display import display

def simple_synth(freq, duration, wave_type="sine",
                 attack=0.01, decay=0.1, sustain=0.7, release=0.1):
    """指定した周波数・波形・エンベロープで音を生成する"""
    # 波形を選択
    wave_funcs = {
        "sine": sine_wave,
        "sawtooth": sawtooth_wave,
        "square": square_wave,
        "triangle": triangle_wave,
    }
    wave_func = wave_funcs[wave_type]
    wave = wave_func(freq, duration)

    # エンベロープを適用
    env = adsr(duration, attack=attack, decay=decay,
               sustain=sustain, release=release)
    return AudioSignal(wave.data * env.data, 44100)

    1. 4種類の波形で C メジャースケール¶

    c_major = [60, 62, 64, 65, 67, 69, 71, 72]
dur_note = 0.5

for wt in ["sine", "sawtooth", "square", "triangle"]:
    parts = []
    for midi_num in c_major:
        freq = note_to_frequency(midi_num)
        sig = simple_synth(freq, dur_note, wave_type=wt)
        parts.append(sig.data)
    scale = AudioSignal(np.concatenate(parts), 44100)
    display(play_sound(scale, f"C メジャースケール（{wt}）"))

    2. 3つのエンベロープで同じメロディ¶

    envelope_presets = {
    "ピアノ風": {"attack": 0.01, "decay": 0.2, "sustain": 0.3, "release": 0.1},
    "バイオリン風": {"attack": 0.3, "decay": 0.1, "sustain": 0.8, "release": 0.2},
    "パーカッション風": {"attack": 0.005, "decay": 0.15, "sustain": 0.0, "release": 0.05},
}

for name, params in envelope_presets.items():
    parts = []
    for midi_num in c_major:
        freq = note_to_frequency(midi_num)
        sig = simple_synth(freq, dur_note, wave_type="sawtooth", **params)
        parts.append(sig.data)
    scale = AudioSignal(np.concatenate(parts), 44100)
    display(play_sound(scale, f"C メジャースケール（{name}）"))

    3. 考察¶

    記述例: 最も「ピアノらしい」と感じたのは、ノコギリ波（sawtooth）+ ピアノ風エンベロープ（A=0.01, D=0.2, S=0.3, R=0.1）の組み合わせ。理由は: - ノコギリ波はすべての整数次倍音を含み、ピアノの弦振動に近い豊かな音色になる - 短いアタック（0.01秒）はハンマーが弦を叩く瞬間に対応する - 比較的長いディケイ（0.2秒）と低いサステイン（0.3）は、ピアノの音が鳴った直後に急速に減衰する特徴に合っている - サイン波は音色が素朴すぎ、矩形波は中空的でクラリネットに近い印象になる

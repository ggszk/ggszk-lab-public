    演習1: ディレイのパラメータ実験¶

    import numpy as np
from IPython.display import display
from audio_lib import sine_wave, adsr, AudioSignal
from audio_lib.notebook import play_sound

def create_test_sound(freq=440, duration=2.0):
    """テスト用ピアノ風の音"""
    wave = sine_wave(freq, duration)
    env = adsr(duration, attack=0.01, decay=0.3, sustain=0.4, release=0.5)
    return AudioSignal(wave.data * env.data, 44100)

def simple_delay(signal, delay_sec=0.3, feedback=0.4, wet_level=0.5,
                 sample_rate=44100):
    """シンプルなディレイエフェクト"""
    delay_samples = int(delay_sec * sample_rate)
    data = signal.data
    output = np.zeros(len(data) + delay_samples * 5)
    buf = np.zeros(delay_samples)
    idx = 0

    for n in range(len(data)):
        delayed = buf[idx]
        buf[idx] = data[n] + feedback * delayed
        output[n] = (1.0 - wet_level) * data[n] + wet_level * delayed
        idx = (idx + 1) % delay_samples

    for n in range(len(data), len(output)):
        delayed = buf[idx]
        buf[idx] = feedback * delayed
        output[n] = wet_level * delayed
        idx = (idx + 1) % delay_samples

    return AudioSignal(np.clip(output, -1.0, 1.0), sample_rate)

dry = create_test_sound()

# 1. 短いディレイ: 個々のエコーは聞き取りにくく、音に厚みが出る
wet1 = simple_delay(dry, delay_sec=0.05, feedback=0.3, wet_level=0.5)
display(play_sound(wet1, "短いディレイ（0.05秒）— 音に厚みが出る"))

# 2. 長いディレイ: 明確なエコーが聞こえる
wet2 = simple_delay(dry, delay_sec=0.5, feedback=0.3, wet_level=0.5)
display(play_sound(wet2, "長いディレイ（0.5秒）— 明確なエコー"))

# 3. フィードバック多め: エコーが何度も繰り返される
wet3 = simple_delay(dry, delay_sec=0.3, feedback=0.7, wet_level=0.5)
display(play_sound(wet3, "フィードバック多め（0.7）— 長い残響"))

    解説: - 短いディレイ（50ms 以下）は個々のエコーが知覚できず、音の厚みや空間感として感じられます - 長いディレイ（200ms 以上）は明確にエコーとして聞こえます - フィードバックを上げるとエコーの繰り返し回数が増え、残響が長くなります

    演習2: コムフィルタの遅延時間と周波数特性¶

    import matplotlib.pyplot as plt
from scipy.fft import fft

def simple_comb_filter(signal, delay_sec=0.03, feedback=0.7,
                       sample_rate=44100):
    """シンプルなコムフィルタ"""
    delay_samples = int(delay_sec * sample_rate)
    data = signal.data
    buf = np.zeros(delay_samples)
    idx = 0
    output = np.zeros_like(data)

    for n in range(len(data)):
        delayed = buf[idx]
        buf[idx] = data[n] + feedback * delayed
        output[n] = delayed
        idx = (idx + 1) % delay_samples

    return AudioSignal(output, sample_rate)

# インパルス信号
impulse = np.zeros(44100)
impulse[0] = 1.0
impulse_signal = AudioSignal(impulse, 44100)

delay_times = [0.005, 0.01, 0.02]

fig, axes = plt.subplots(3, 1, figsize=(12, 10))

for ax, delay_sec in zip(axes, delay_times):
    comb_out = simple_comb_filter(impulse_signal, delay_sec=delay_sec,
                                  feedback=0.7)
    spectrum = np.abs(fft(comb_out.data))
    freqs = np.arange(len(spectrum)) / len(spectrum) * 44100

    expected_spacing = 1.0 / delay_sec
    ax.plot(freqs[:5000], 20 * np.log10(spectrum[:5000] + 1e-10))
    ax.set_title(f"遅延 {delay_sec}秒 → 山の間隔 = 1/{delay_sec} = {expected_spacing:.0f} Hz")
    ax.set_xlabel("周波数 (Hz)")
    ax.set_ylabel("振幅 (dB)")
    ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

    解説: - 遅延 0.005 秒 → 山の間隔 200 Hz - 遅延 0.01 秒 → 山の間隔 100 Hz - 遅延 0.02 秒 → 山の間隔 50 Hz

    遅延時間が短いほど山の間隔が広くなり、長いほど密になります。山の間隔は 1 / delay_sec Hz に一致します。

    演習3: コンプレッサーの閾値と圧縮比¶

    from audio_lib import note_to_frequency

def simple_compressor(signal, threshold=0.5, ratio=4.0,
                      attack=0.01, release=0.1, sample_rate=44100):
    """シンプルなコンプレッサー"""
    data = signal.data
    output = np.zeros_like(data)
    envelope = 0.0
    attack_coeff = np.exp(-1.0 / (attack * sample_rate))
    release_coeff = np.exp(-1.0 / (release * sample_rate))

    for n in range(len(data)):
        current_level = abs(data[n])
        if current_level > envelope:
            envelope += (current_level - envelope) * (1 - attack_coeff)
        else:
            envelope += (current_level - envelope) * (1 - release_coeff)

        if envelope > threshold:
            excess = envelope - threshold
            target = threshold + excess / ratio
            gain = target / envelope
        else:
            gain = 1.0
        output[n] = data[n] * gain

    return AudioSignal(output, sample_rate)

# 音量がバラバラのメロディ
notes = [60, 64, 67, 72, 67, 64]
amplitudes = [0.3, 0.9, 0.5, 1.0, 0.4, 0.8]

parts = []
for midi_num, amp in zip(notes, amplitudes):
    freq = note_to_frequency(midi_num)
    wave = sine_wave(freq, 0.5)
    env = adsr(0.5, attack=0.01, decay=0.1, sustain=0.6, release=0.1)
    shaped = AudioSignal(wave.data * env.data * amp, 44100)
    parts.append(shaped.data)

dynamic = AudioSignal(np.concatenate(parts), 44100)

# 軽い圧縮
light = simple_compressor(dynamic, threshold=0.6, ratio=2.0)
# 強い圧縮
heavy = simple_compressor(dynamic, threshold=0.3, ratio=8.0)

fig, axes = plt.subplots(3, 1, figsize=(12, 8), sharex=True)
t = np.arange(len(dynamic.data)) / 44100

axes[0].plot(t, dynamic.data, alpha=0.7)
axes[0].set_title("元の音")
axes[0].set_ylabel("振幅")
axes[0].set_ylim(-1.1, 1.1)
axes[0].grid(True, alpha=0.3)

axes[1].plot(t, light.data, alpha=0.7, color='green')
axes[1].set_title("軽い圧縮（threshold=0.6, ratio=2.0）")
axes[1].set_ylabel("振幅")
axes[1].set_ylim(-1.1, 1.1)
axes[1].grid(True, alpha=0.3)

axes[2].plot(t, heavy.data, alpha=0.7, color='orange')
axes[2].set_title("強い圧縮（threshold=0.3, ratio=8.0）")
axes[2].set_xlabel("時間 (秒)")
axes[2].set_ylabel("振幅")
axes[2].set_ylim(-1.1, 1.1)
axes[2].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

    解説: - 軽い圧縮: 大きな音がやや抑えられる程度で、自然なダイナミクスが残ります - 強い圧縮: ほぼすべての音が同じ音量になり、均一な印象になります。音量差がなくなる代わりに、音楽的な抑揚も失われます

    演習4: ピンポンディレイ¶

    def ping_pong_delay(signal, delay_sec=0.3, feedback=0.4, wet_level=0.5,
                    sample_rate=44100):
    """ピンポンディレイ（ステレオ）"""
    delay_samples = int(delay_sec * sample_rate)
    data = signal.data
    total_len = len(data) + delay_samples * 8
    left = np.zeros(total_len)
    right = np.zeros(total_len)
    buf = np.zeros(delay_samples)
    idx = 0
    echo_count = 0  # エコーの回数

    for n in range(total_len):
        # 入力信号の範囲内なら入力を取得
        x = data[n] if n < len(data) else 0.0

        delayed = buf[idx]

        if n < len(data):
            buf[idx] = x + feedback * delayed
        else:
            buf[idx] = feedback * delayed

        # バッファが一周するたびにエコー回数を更新
        if idx == 0 and n > 0:
            echo_count += 1

        # 元の音は両チャンネルに、エコーは交互に
        dry_signal = (1.0 - wet_level) * x
        wet_signal = wet_level * delayed

        if echo_count % 2 == 0:
            left[n] = dry_signal + wet_signal
            right[n] = dry_signal
        else:
            left[n] = dry_signal
            right[n] = dry_signal + wet_signal

        idx = (idx + 1) % delay_samples

    # ステレオ配列（2チャンネル）
    stereo = np.column_stack([
        np.clip(left, -1.0, 1.0),
        np.clip(right, -1.0, 1.0)
    ])

    return stereo

dry = create_test_sound()
stereo_out = ping_pong_delay(dry, delay_sec=0.3, feedback=0.5, wet_level=0.5)
print(f"出力の形状: {stereo_out.shape}")  # (サンプル数, 2)

    解説: ピンポンディレイでは、エコーが左右のチャンネルに交互に振り分けられるため、ヘッドフォンで聴くと音が左右に跳ねるように聞こえます。echo_count でバッファの周回数を追跡し、奇数/偶数で振り分けています。

    演習5: エフェクトチェーンの順序比較¶

    from pedalboard import Pedalboard, Chorus, Delay, Reverb
from audio_lib.notebook import apply_effect

dry = create_test_sound(duration=2.0)

# パターン1: コーラス → ディレイ → リバーブ
chain1 = Pedalboard([
    Chorus(rate_hz=2.0, depth=0.25, mix=0.3),
    Delay(delay_seconds=0.2, feedback=0.3, mix=0.3),
    Reverb(room_size=0.7, wet_level=0.4),
])
display(play_sound(apply_effect(dry, chain1),
                   "コーラス → ディレイ → リバーブ"))

# パターン2: リバーブ → コーラス → ディレイ
chain2 = Pedalboard([
    Reverb(room_size=0.7, wet_level=0.4),
    Chorus(rate_hz=2.0, depth=0.25, mix=0.3),
    Delay(delay_seconds=0.2, feedback=0.3, mix=0.3),
])
display(play_sound(apply_effect(dry, chain2),
                   "リバーブ → コーラス → ディレイ"))

# パターン3: ディレイ → リバーブ → コーラス
chain3 = Pedalboard([
    Delay(delay_seconds=0.2, feedback=0.3, mix=0.3),
    Reverb(room_size=0.7, wet_level=0.4),
    Chorus(rate_hz=2.0, depth=0.25, mix=0.3),
])
display(play_sound(apply_effect(dry, chain3),
                   "ディレイ → リバーブ → コーラス"))

    解説: - パターン1（コーラス → ディレイ → リバーブ）: 厚みのある音がエコーし、全体が残響に包まれる。最も自然で定番の順序 - パターン2（リバーブ → コーラス → ディレイ）: 残響音が揺れてエコーするため、独特の浮遊感がある - パターン3（ディレイ → リバーブ → コーラス）: エコーが残響に溶け込んだ後に揺れるため、全体がぼんやりとした印象

    演習6（発展）: フランジャーの実装¶

    def flanger(signal, rate=0.5, min_delay=0.001, max_delay=0.005,
            feedback=0.7, wet_level=0.5, sample_rate=44100):
    """フランジャーエフェクト"""
    data = signal.data
    max_delay_samples = int(max_delay * sample_rate) + 2
    buf = np.zeros(max_delay_samples)
    buf_idx = 0
    output = np.zeros_like(data)
    phase = 0.0

    for n in range(len(data)):
        # LFO で遅延時間を変調
        lfo = np.sin(2 * np.pi * phase)
        delay_sec = min_delay + (max_delay - min_delay) * (lfo + 1) / 2
        delay_samples = delay_sec * sample_rate

        # 線形補間で小数サンプル遅延を処理
        delay_int = int(np.floor(delay_samples))
        frac = delay_samples - delay_int
        idx0 = (buf_idx - delay_int) % max_delay_samples
        idx1 = (buf_idx - delay_int - 1) % max_delay_samples
        delayed = (1.0 - frac) * buf[idx0] + frac * buf[idx1]

        # フィードバック付きでバッファに書き込み
        buf[buf_idx] = data[n] + feedback * delayed

        # ドライ + ウェットを出力
        output[n] = (1.0 - wet_level) * data[n] + wet_level * delayed

        buf_idx = (buf_idx + 1) % max_delay_samples
        phase += rate / sample_rate
        if phase >= 1.0:
            phase -= 1.0

    return AudioSignal(np.clip(output, -1.0, 1.0), sample_rate)

dry = create_test_sound(duration=3.0)

# フランジャー
flanged = flanger(dry, rate=0.3, feedback=0.7, wet_level=0.5)
display(play_sound(flanged, "フランジャー（rate=0.3, feedback=0.7）"))

# コーラスとの比較
chorus_out = apply_effect(dry, Chorus(rate_hz=0.3, depth=0.25, mix=0.5))
display(play_sound(chorus_out, "コーラス（比較用）"))

    解説: - フランジャーはコーラスと構造が似ていますが、遅延時間が短い（1〜5ms）ためにコムフィルタ効果が生じ、金属的な「ジェット機のような」音になります - コーラスは遅延時間が長め（5〜20ms）でフィードバックがないため、柔らかい厚みのある音になります - フィードバックの有無と遅延時間の長さが、両者の音の違いを決める主な要因です
